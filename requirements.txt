# Python 3.12. SafeChain requires it, and the pipeline is run on it.

# Install from the internal index:
#     pip install -r requirements.txt

# Model calls. SafeChain owns authentication, token refresh and the request body each model
# expects, and returns a LangChain chat model. langchain-core comes with it and is listed because
# this package imports it directly to build prompts; nothing here talks HTTP to a model.
safechain
langchain-core

# Core pipeline.
openpyxl
python-dotenv
pyyaml

# Document ingestion. Each parser is imported inside the reader that needs it, so a package
# missing from an internal mirror disables one format rather than the whole tool.
pypdf
python-docx
python-pptx
Pillow

# Local web interface. Flask bundles Jinja2 and Werkzeug; nothing else is needed, and there is
# no Node or npm step -- the interface is server-rendered HTML with hand-written CSS.
#
# Deliberately unpinned, and the code stays on the Flask 1.x interface so that whatever version
# an internal mirror already carries will run it. tests/test_flask_compatibility.py holds that
# line: the development environment usually has a newer Flask than the machine this runs on, so
# a 2.0-only convenience passes every test here and fails where it matters.
flask
